cmake_minimum_required(VERSION 4.2)


project(WyvernServer LANGUAGES C CXX)

set(CMAKE_MODULE_PATH ${PROJECT_SOURCE_DIR}/../cmake)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(DetectSystemParams)
DetectSystemParams()

cmake_policy(SET CMP0167 OLD)
set(WYVERN_RESOLVED_CORE_DIR "${CMAKE_SOURCE_DIR}/Core")

if(NOT WYVERN_RESOLVED_CORE_DIR)
  message(FATAL_ERROR
    "Core was not found. Set WYVERN_CORE_DIR to external Core path "
    "or keep Core as ../Core (preferred) / Core (fallback)."
  )
endif()

message(STATUS "Using Core from: ${WYVERN_RESOLVED_CORE_DIR}")
add_subdirectory("${WYVERN_RESOLVED_CORE_DIR}" "${CMAKE_BINARY_DIR}/_deps/wyvern_core")

option(WYVERN_BUILD_P2P_MESSENGER "Build P2P messenger target" ON)
option(WYVERN_BUILD_P2P_TESTS "Build P2P transport unit and loopback tests" ON)

if(WYVERN_BUILD_P2P_MESSENGER)
  find_package(Boost REQUIRED COMPONENTS system thread)

  # === Embedded Python: no system Python required ===
  include(BootstrapPythonSDK)
  BootstrapPythonSDK()

  include(BootstrapPythonRuntime)
  BootstrapPythonRuntime()

  # Create IMPORTED target for the embedded Python SDK.
  if(NOT TARGET Python3::Python)
    add_library(Python3::Python SHARED IMPORTED)
    if(WYVERN_SYSTEM_FAMILY_WINDOWS)
      set_target_properties(Python3::Python PROPERTIES
        IMPORTED_IMPLIB   "${PYTHON_SDK_ROOT}/libs/python312.lib"
        IMPORTED_LOCATION "${PYTHON_SDK_ROOT}/python312.dll"
        INTERFACE_INCLUDE_DIRECTORIES "${PYTHON_SDK_ROOT}/include"
      )
    else()
      set_target_properties(Python3::Python PROPERTIES
        IMPORTED_LOCATION "${PYTHON_SDK_ROOT}/lib/libpython3.12.so"
        INTERFACE_INCLUDE_DIRECTORIES "${PYTHON_SDK_ROOT}/include/python3.12"
      )
    endif()
  endif()

  # pybind11 via FetchContent (header-only, no vcpkg Python dependency).
  # Sources are cached in EXTERNALS (persistent, one download ever).
  # Subbuild stays inside the build tree (generator-specific, regenerated on fresh).
  include(FetchContent)
  set(_pybind11_src_dir "${CMAKE_SOURCE_DIR}/../EXTERNALS/_fetchcontent_cache/pybind11-src")
  FetchContent_Declare(pybind11
    URL      https://github.com/pybind/pybind11/archive/refs/tags/v2.13.6.tar.gz
    URL_HASH SHA256=e08cb87f4773da97fa7b5f035de8763abc656d87d5773e62f6da0587d1f0ec20
    SOURCE_DIR        "${_pybind11_src_dir}"
    BINARY_DIR        "${CMAKE_BINARY_DIR}/_deps/pybind11-build"
    SUBBUILD_DIR      "${CMAKE_BINARY_DIR}/_deps/pybind11-subbuild"
  )
  set(PYBIND11_FINDPYTHON OFF CACHE BOOL "" FORCE)
  set(PYBIND11_NOPYTHON   ON  CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(pybind11)

  add_executable(p2p_messenger
    Application/platform/P2pMessengerMain.cpp
    Application/platform/P2pPlatformModules.cpp
    Application/modules/p2p/transport/P2pConnectionModule.cpp
    Application/modules/p2p/crypto/PythonRuntimeModule.cpp
    Application/modules/p2p/crypto/MeshCryptoModule.cpp
    Application/modules/p2p/db/MeshNodeDbModule.cpp
    Application/modules/p2p/app/P2pMessengerModule.cpp
    Application/modules/p2p/relay/RelaySignalingModule.cpp
    Application/modules/dataStorage/InMemoryRelayStore.cpp
  )

  target_include_directories(p2p_messenger
    PRIVATE
      ${CMAKE_SOURCE_DIR}/Application/platform
      ${CMAKE_SOURCE_DIR}/Application/modules
      ${CMAKE_SOURCE_DIR}/Application/modules/p2p
      ${CMAKE_SOURCE_DIR}/Application/modules/Transport
      ${CMAKE_SOURCE_DIR}/Application/relay
  )

  target_link_libraries(p2p_messenger
    PRIVATE
      WyvernCore
      Boost::system
      Boost::thread
      Python3::Python
      pybind11::embed
  )

  if(WIN32)
    target_compile_definitions(p2p_messenger PRIVATE
      _WIN32_WINNT=0x0A00
      WIN32_LEAN_AND_MEAN
      NOMINMAX
    )
    target_link_libraries(p2p_messenger PRIVATE ws2_32 mswsock)
  endif()
  target_compile_definitions(p2p_messenger PRIVATE
    # Debug — папка в EXTERNALS (разработка)
    $<$<CONFIG:Debug>:PYTHON_HOME_PATH="${CMAKE_SOURCE_DIR}/../EXTERNALS/python">

    # Release — папка рядом с exe (деплой)
    $<$<CONFIG:Release>:PYTHON_HOME_PATH="$<TARGET_FILE_DIR:p2p_messenger>/python">
  )

    # Опционально: для отладки в Visual Studio (только Debug)
  set_target_properties(p2p_messenger PROPERTIES
      VS_DEBUGGER_ENVIRONMENT "$<$<CONFIG:Debug>:PYTHONHOME=${CMAKE_SOURCE_DIR}/../EXTERNALS/python>"
  )

    )
    target_link_libraries(p2p_tests PRIVATE ws2_32 mswsock)
