cmake_minimum_required(VERSION 4.2)


project(WyvernServer LANGUAGES C CXX)

set(CMAKE_MODULE_PATH ${PROJECT_SOURCE_DIR}/../cmake)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(DetectSystemParams)
DetectSystemParams()

cmake_policy(SET CMP0167 OLD)

option(WYVERN_BUILD_P2P_MESSENGER "Build P2P messenger target" ON)

if(WYVERN_BUILD_P2P_MESSENGER)
  find_package(Boost REQUIRED COMPONENTS system thread)
  find_package(LibDataChannel CONFIG REQUIRED)

  add_executable(p2p_messenger
    Application/main/Loader.cpp
    Application/main/Runtime.cpp
   )

  target_include_directories(p2p_messenger
    PRIVATE
      ${CMAKE_SOURCE_DIR}/Application/main
  )

  target_link_libraries(p2p_messenger
    PRIVATE
      Boost::system
      Boost::thread
      LibDataChannel::LibDataChannel
  )
  if(WIN32)
    target_compile_definitions(p2p_messenger PRIVATE
      _WIN32_WINNT=0x0A00
      WIN32_LEAN_AND_MEAN
      NOMINMAX
    )
    target_link_libraries(p2p_messenger PRIVATE ws2_32 mswsock)
  endif()
endif()
