cmake_minimum_required(VERSION 3.20)
project(WyvernServer LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(POLICY CMP0141)
  cmake_policy(SET CMP0141 NEW)
endif()
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 OLD)
endif()

option(WYVERN_BUILD_FOUNDATION "Build Wyvern foundation server target" ON)

set(WYVERN_CORE_DIR "" CACHE PATH "Path to external Core directory")
set(_wyvern_core_candidates "")
if(WYVERN_CORE_DIR)
  list(APPEND _wyvern_core_candidates "${WYVERN_CORE_DIR}")
endif()
list(APPEND _wyvern_core_candidates
  "${CMAKE_SOURCE_DIR}/../Core"
  "${CMAKE_SOURCE_DIR}/Core"
)

set(WYVERN_RESOLVED_CORE_DIR "")
foreach(_candidate IN LISTS _wyvern_core_candidates)
  if(EXISTS "${_candidate}/CmakeLists.txt" OR EXISTS "${_candidate}/CMakeLists.txt")
    set(WYVERN_RESOLVED_CORE_DIR "${_candidate}")
    break()
  endif()
endforeach()

if(NOT WYVERN_RESOLVED_CORE_DIR)
  message(FATAL_ERROR
    "Core was not found. Set WYVERN_CORE_DIR to external Core path "
    "or keep Core as ../Core (preferred) / Core (fallback)."
  )
endif()

message(STATUS "Using Core from: ${WYVERN_RESOLVED_CORE_DIR}")
add_subdirectory("${WYVERN_RESOLVED_CORE_DIR}" "${CMAKE_BINARY_DIR}/_deps/wyvern_core")

option(WYVERN_BUILD_P2P_MESSENGER "Build P2P messenger target" ON)

if(WYVERN_BUILD_FOUNDATION)
  find_package(Boost REQUIRED COMPONENTS system thread)

  add_executable(wyvern_server_foundation
    Application/platform/ServerFoundationMain.cpp
    Application/platform/PlatformModules.cpp
    Application/modules/httpServerLegasy/FileCache.cpp
    Application/modules/httpServerLegasy/RequestHandler.cpp
    Application/modules/Transport/udp/udpLoopbackTest.cpp
  )

  target_include_directories(wyvern_server_foundation
    PRIVATE
      ${CMAKE_SOURCE_DIR}/Application/platform
      ${CMAKE_SOURCE_DIR}/Application/modules/httpServerLegasy
      ${CMAKE_SOURCE_DIR}/Application/modules/httpServerLegasy/addons
      ${CMAKE_SOURCE_DIR}/Application/modules/httpServerLegasy/support
      ${CMAKE_SOURCE_DIR}/Application/modules/Transport
  )

  target_link_libraries(wyvern_server_foundation
    PRIVATE
      WyvernCore
      Boost::system
      Boost::thread
  )

  if(WIN32)
    target_compile_definitions(wyvern_server_foundation PRIVATE _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(wyvern_server_foundation PRIVATE ws2_32 mswsock)
  endif()
endif()

if(WYVERN_BUILD_P2P_MESSENGER)
  find_package(Boost REQUIRED COMPONENTS system thread)

  # On Windows, prefer the system (registry) Python over vcpkg's debug Python.
  # vcpkg's debug Python requires extension modules with _d suffix (_rust_d.pyd etc.)
  # which third-party packages do not ship. Using the release Python in all
  # build configs avoids this incompatibility.
  if(WIN32)
    set(Python3_FIND_REGISTRY FIRST)
  endif()
  # Do not append debug postfix when searching for the Python library.
  set(Python3_DEBUG_POSTFIX "")
  find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
  find_package(pybind11 CONFIG REQUIRED)

  add_executable(p2p_messenger
    Application/platform/P2pMessengerMain.cpp
    Application/platform/P2pPlatformModules.cpp
    Application/modules/p2p/transport/P2pConnectionModule.cpp
    Application/modules/p2p/crypto/PythonRuntimeModule.cpp
    Application/modules/p2p/crypto/MeshCryptoModule.cpp
    Application/modules/p2p/db/MeshNodeDbModule.cpp
    Application/modules/p2p/app/P2pMessengerModule.cpp
  )

  target_include_directories(p2p_messenger
    PRIVATE
      ${CMAKE_SOURCE_DIR}/Application/platform
      ${CMAKE_SOURCE_DIR}/Application/modules
      ${CMAKE_SOURCE_DIR}/Application/modules/p2p
      ${CMAKE_SOURCE_DIR}/Application/modules/Transport
  )

  target_link_libraries(p2p_messenger
    PRIVATE
      WyvernCore
      Boost::system
      Boost::thread
      Python3::Python
      pybind11::embed
  )

  if(WIN32)
    target_compile_definitions(p2p_messenger PRIVATE
      _WIN32_WINNT=0x0A00
      WIN32_LEAN_AND_MEAN
      NOMINMAX
    )
    target_link_libraries(p2p_messenger PRIVATE ws2_32 mswsock)
    # Force the release Python library even in Debug C++ builds.
    # The debug Python DLL requires _d.pyd extensions; third-party packages
    # only ship release .pyd files, so the debug Python will fail to import them.
    if(Python3_LIBRARY_RELEASE)
      target_link_libraries(p2p_messenger PRIVATE ${Python3_LIBRARY_RELEASE})
    endif()
  endif()
endif()
